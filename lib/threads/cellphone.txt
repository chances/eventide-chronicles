// Cell phone thread. Reads keys, drives cell-phone.cleo (0B40 to 0B42), and draws the phone with sprites.
// Needs lib/phone.txt and lib/threads/cellphone-layout.txt included first. The design is in plugins/cell-phone/README.md.
// Screen units are the 640 by 448 space of the sprite commands. PHONE_TOP puts the last key row 13 mockup px
// above the bottom edge: 448 - (378 + 13) * PHONE_SY.
const PHONE_SX = 0.45
const PHONE_SY = 0.6
const PHONE_LEFT = 520.0
const PHONE_TOP = 213.4
const PHONE_SLIDE = 260.0
const PHONE_SLIDE_MS = 550
const PHONE_TAXI = 5558294
const PHONE_PIZZA = 5557492
const TAXI_SPAWN_DISTANCE = 60.0
const TAXI_ARRIVE_D2 = 64.0

:CellPhone
set_deatharrest_state 0
script_name 'CPHONE'

// The phone is owned for good once the pickup is collected
while $PhoneOwned == 0
    wait 250
    if
        Pickup.HasBeenCollected($CellPhone)
    then
        $PhoneOwned = 1
    end
end

Txd.LoadDictionary('phone')
CellPhone_Sprites()
timera = 0
$PhoneUpPrev = 0
$PhonePrev = -1
$PhoneControlOff = 0

while true
    wait 0
    CellPhone_Input()
    CellPhone_Tick()
    CellPhone_Events()
    CellPhone_Draw()
end

// Reacts to the plugin's events and runs the dispatches a dialed number started.
:CellPhone_Events
if
    $PhoneEvent == 1
then
    CellPhone_DialDone()
end
CellPhone_TaxiTick()
Pizza_DialTick()
return

// A dialed number finished ringing. The plugin knows nothing about numbers, so each number's handler answers
// with 0B42: 1 connected, 2 busy, 3 call over. A number with no handler gets the busy tone.
:CellPhone_DialDone
$PhoneDialed = 0
if
    $PhoneNumber == PHONE_TAXI
then
    $PhoneDialed = 1
    0B42: 1
    CellPhone_Taxi()
    0B42: 3
end
if
    $PhoneNumber == PHONE_PIZZA
then
    $PhoneDialed = 1
    0B42: 1
    Pizza_Dial()
    0B42: 3
end
if
    $PhoneDialed == 0
then
    0B42: 2
end
return

// Dispatches a taxi. A second call while one is on its way connects and changes nothing.
:CellPhone_Taxi
if
    $TaxiStage == 0
then
    $TaxiStage = 1
    $TaxiDir = 0
end
return

// One step per frame. Stages: 1 request models, 2 wait for them, 3 find a road node out of camera view and
// spawn, 4 drive to the player, 5 park.
:CellPhone_TaxiTick
if
    $TaxiStage == 0
then
    return
end
if
    $TaxiStage == 1
then
    Streaming.RequestModel(#TAXI)
    Streaming.RequestModel(#HMOST)
    $TaxiStage = 2
    return
end
if
    $TaxiStage == 2
then
    if and
        Streaming.HasModelLoaded(#TAXI)
        Streaming.HasModelLoaded(#HMOST)
    then
        $TaxiStage = 3
    end
    return
end
if
    $TaxiStage == 3
then
    // Try one side of the player per frame (east, west, north, south) until its road node is off screen
    $TaxiPX, $TaxiPY, $TaxiPZ = Char.GetCoordinates($scplayer)
    $TaxiX = $TaxiPX
    $TaxiY = $TaxiPY
    if
        $TaxiDir == 0
    then
        $TaxiX += TAXI_SPAWN_DISTANCE
    end
    if
        $TaxiDir == 1
    then
        $TaxiX -= TAXI_SPAWN_DISTANCE
    end
    if
        $TaxiDir == 2
    then
        $TaxiY += TAXI_SPAWN_DISTANCE
    end
    if
        $TaxiDir == 3
    then
        $TaxiY -= TAXI_SPAWN_DISTANCE
    end
    $TaxiNX, $TaxiNY, $TaxiNZ = Path.GetClosestCarNode($TaxiX, $TaxiY, $TaxiPZ)
    $TaxiDir += 1
    if
        $TaxiDir > 3
    then
        $TaxiDir = 0
    end
    if
        not Camera.IsPointOnScreen($TaxiNX, $TaxiNY, $TaxiNZ, 5.0)
    then
        $TaxiCar = Car.Create(#TAXI, $TaxiNX, $TaxiNY, $TaxiNZ)
        $TaxiDriver = Char.CreateInsideCar($TaxiCar, PedType.CivMale, #HMOST)
        Car.SetDrivingStyle($TaxiCar, DrivingMode.AvoidCars)
        Car.SetCruiseSpeed($TaxiCar, 20.0)
        Streaming.MarkModelAsNoLongerNeeded(#TAXI)
        Streaming.MarkModelAsNoLongerNeeded(#HMOST)
        $TaxiFrames = 0
        $TaxiStage = 4
    end
    return
end
if
    $TaxiStage == 4
then
    if
        Car.IsDead($TaxiCar)
    then
        Car.MarkAsNoLongerNeeded($TaxiCar)
        Char.MarkAsNoLongerNeeded($TaxiDriver)
        $TaxiStage = 0
        return
    end
    $TaxiPX, $TaxiPY, $TaxiPZ = Char.GetCoordinates($scplayer)
    $TaxiNX, $TaxiNY, $TaxiNZ = Car.GetCoordinates($TaxiCar)
    $TaxiDX = $TaxiNX
    $TaxiDX -= $TaxiPX
    $TaxiDY = $TaxiNY
    $TaxiDY -= $TaxiPY
    $TaxiDX *= $TaxiDX
    $TaxiDY *= $TaxiDY
    $TaxiD2 = $TaxiDX
    $TaxiD2 += $TaxiDY
    if
        $TaxiD2 < TAXI_ARRIVE_D2
    then
        // The car stays owned while it waits: marking it no longer needed hands it to the traffic system,
        // which clears the stop order and drives it off.
        Car.SetMission($TaxiCar, CarMission.StopForever)
        $TaxiStage = 5
        return
    end
    // Slow down on the approach so the stop order catches the car near the player
    if
        $TaxiD2 < 900.0
    then
        Car.SetCruiseSpeed($TaxiCar, 8.0)
    end
    // The order is repeated about once a second, because the player moves
    if
        $TaxiFrames == 0
    then
        Car.GotoCoordinatesAccurate($TaxiCar, $TaxiPX, $TaxiPY, $TaxiPZ)
        $TaxiFrames = 60
    end
    $TaxiFrames -= 1
    return
end
// Stage 5: parked. The taxi is released when it is wrecked or the player is 100 m away.
if
    Car.IsDead($TaxiCar)
then
    Car.MarkAsNoLongerNeeded($TaxiCar)
    Char.MarkAsNoLongerNeeded($TaxiDriver)
    $TaxiStage = 0
    return
end
$TaxiPX, $TaxiPY, $TaxiPZ = Char.GetCoordinates($scplayer)
$TaxiNX, $TaxiNY, $TaxiNZ = Car.GetCoordinates($TaxiCar)
$TaxiDX = $TaxiNX
$TaxiDX -= $TaxiPX
$TaxiDY = $TaxiNY
$TaxiDY -= $TaxiPY
$TaxiDX *= $TaxiDX
$TaxiDY *= $TaxiDY
$TaxiD2 = $TaxiDX
$TaxiD2 += $TaxiDY
if
    $TaxiD2 > 10000.0
then
    Car.MarkAsNoLongerNeeded($TaxiCar)
    Char.MarkAsNoLongerNeeded($TaxiDriver)
    $TaxiStage = 0
end
return

$PhoneSend = $PhoneKey

    $PhoneKey == $PhonePrev
then
    $PhoneSend = -1
end
$PhonePrev = $PhoneKey
// A script can press a key for the phone: it stores the key code plus one in $PhoneInject, and 0 means none.
// The probe uses it for scripted calls. The pressed look does not follow an injected key.
if
    $PhoneInject > 0
then
    $PhoneSend = $PhoneInject
    $PhoneSend -= 1
    $PhoneInject = 0
end
return

// Runs the plugin for one frame, then mirrors its state into the globals other scripts read.
:CellPhone_Tick
$PhoneBlocked = 0
if
    Player.IsInAnyCar($PLAYER1)
then
    $PhoneBlocked = 1
end
if
    not Player.IsPlaying($PLAYER1)

    $PhoneBlocked = 1

    $onCutscene == 1

0B40: $PhoneSend $PhoneBlocked $PhoneState $PhoneLit $PhoneAnim $PhoneCount $PhoneNumber $PhoneEvent

// Up (shown) while raised, dialing, ringing in, or in a call. The slide restarts when that flips.
$PhoneUp = 0
if or
    $PhoneState == 1
    $PhoneState == 2
    $PhoneState == 3
    $PhoneState == 4
    $PhoneState == 5
then
    $PhoneUp = 1
end
if
    not $PhoneUp == $PhoneUpPrev
then
    timera = 0
    $PhoneUpPrev = $PhoneUp
end

// The player cannot move while the player is using the phone. A ringing phone leaves control alone.
$PhoneBusy = 0
if or
    $PhoneState == 1
    $PhoneState == 2
    $PhoneState == 3
    $PhoneState == 5
then
    $PhoneBusy = 1
end
if
    not $PhoneBusy == $PhoneControlOff
then
    if
        $PhoneBusy == 1
    then
        Player.SetControl($PLAYER1, False)
    else
        Player.SetControl($PLAYER1, True)
    end
    $PhoneControlOff = $PhoneBusy
end

// An incoming call: the caller's number flashes and a help text names the keys. $PhoneCaller holds the
// caller's seven digits, set by whoever calls cell_phone_ring_incoming.
if
    not $PhoneState == $PhoneStatePrev
then
    if
        $PhoneState == 4
    then
        timerb = 0
        $PhoneFlash = 1
        Text.PrintHelpForever('PHRING')
    end
    if
        $PhoneStatePrev == 4
    then
        Text.ClearHelp()
    end
    // A call is timed from the moment it is dialed, or from the moment it is answered when it rang in. A dialed
    // call keeps its clock when it connects.
    if
        $PhoneState == 2
    then
        timerb = 0
        $CallSeconds = 0
    end
    if
        $PhoneState == 5
    then
        if
            $PhoneStatePrev == 4
        then
            timerb = 0
            $CallSeconds = 0
        end
    end
    $PhoneStatePrev = $PhoneState
end

// Whole seconds of the call (dialing, dialed, or connected), counted from timerb with its remainder kept, so
// the clock does not drift
if or
    $PhoneState == 2
    $PhoneState == 3
    $PhoneState == 5
then
    if
        timerb >= 1000
    then
        $CallMs = timerb
        $CallMs -= 1000
        timerb = $CallMs
        $CallSeconds += 1
    end
end
return

    if

        then
            $PhoneFlash = 0

            $PhoneFlash = 1

// The LCD shows the number or the call timer, never both. A dialed call keeps its number on screen for the
// first half second, then the timer replaces it. A call with no number on screen, such as an answered one,
// shows the timer at once. The plugin starts the ringback at the same moment (kRingbackDelayMs in
// plugins/cell-phone/tones.h), so the two 500 ms values change together.
$ShowTimer = 0

    $PhoneState == 2
    $PhoneState == 3
    $PhoneState == 5
then
    $ShowTimer = 1
    if
        $PhoneCount > 0
    then
        if
            $CallSeconds == 0
        then
            if
                timerb < 500
            then
                $ShowTimer = 0
            end
        end
    end
end

    $ShowTimer == 1

    CellPhone_Timer()
else
    CellPhone_Digits()

// The typed digits, left aligned on the LCD, with a dash after the third. Leading zeros stay because the
// plugin reports how many digits were typed. The game has no monospaced font and its text wrapped the digits,
// so each glyph is a sprite of the mockup's Courier New (slots 50 to 59 for 0 to 9, 60 for the dash). The
// glyphs are white, and the sprite color tints them: Ember when lit, the muted color at 18% alpha when not.
:CellPhone_Digits
$TCount = $PhoneCount
$TRest = $PhoneNumber
if
    $PhoneState == 4
then
    // Ringing in: the caller's seven digits, flashing
    if
        $PhoneFlash == 0
    then
        return
    end
    $TCount = 7
    $TRest = $PhoneCaller
end
if
    $TCount == 0
then
    return
end
CellPhone_GlyphColor()
CellPhone_GlyphMetrics()
$TextX = PH_GLYPH_LEFT
$TextX += PH_GLYPH_CX
$TextX *= PHONE_SX
$TextX += PHONE_LEFT
$TX = $TextX
$TIndex = 0
while $TIndex < $TCount
    if
        $TIndex == 3
    then
        Hud.DrawSprite(60, $TX, $TextY, $TW, $TH, $GR, $GG, $GB, $GA)
        $TX += $TAdv
    end

    // Place value of this digit: 10 to the power of (count - 1 - index)
    $TExp = $TCount
    $TExp -= 1
    $TExp -= $TIndex
    $TPlace = 1
    if
        $TExp == 1
    then
        $TPlace = 10
    end
    if
        $TExp == 2
    then
        $TPlace = 100
    end
    if
        $TExp == 3
    then
        $TPlace = 1000
    end
    if
        $TExp == 4
    then
        $TPlace = 10000
    end
    if
        $TExp == 5
    then
        $TPlace = 100000
    end
    if
        $TExp == 6
    then
        $TPlace = 1000000
    end
    $TDigit = 0
    while $TRest >= $TPlace
        $TRest -= $TPlace
        $TDigit += 1
    end

    $PSlot = 50
    $PSlot += $TDigit
    Hud.DrawSprite($PSlot, $TX, $TextY, $TW, $TH, $GR, $GG, $GB, $GA)
    $TX += $TAdv
    $TIndex += 1
end
return

// The elapsed time of a call (dialing, dialed, or connected) as m:ss, right aligned on the LCD (slot 61 is the colon). The minutes
// take a second digit from 10 and stop at 99. $CallSeconds is counted in CellPhone_Tick.
:CellPhone_Timer
CellPhone_GlyphColor()
CellPhone_GlyphMetrics()
$TMin = 0
$TSec = $CallSeconds
while $TSec >= 60
    $TSec -= 60
    $TMin += 1
end
if
    $TMin > 99
then
    $TMin = 99
end
$TTens = 0
while $TSec >= 10
    $TSec -= 10
    $TTens += 1
end
$TM10 = 0
while $TMin >= 10
    $TMin -= 10
    $TM10 += 1
end

// Four cells, or five when the minutes have two digits. The row ends at the LCD's padding edge.
$TWide = PH_GLYPH_ADV
$TWide *= 4.0
if
    $TM10 > 0
then
    $TWide = PH_GLYPH_ADV
    $TWide *= 5.0
end
$TextX = PH_LCDBOX_X
$TextX += PH_LCDBOX_W
$TextX -= 14.0
$TextX -= $TWide
$TextX += PH_GLYPH_CX
$TextX *= PHONE_SX
$TextX += PHONE_LEFT
$TX = $TextX
if
    $TM10 > 0
then
    $PSlot = 50
    $PSlot += $TM10
    CellPhone_Glyph()
end
$PSlot = 50
$PSlot += $TMin
CellPhone_Glyph()
$PSlot = 61
CellPhone_Glyph()
$PSlot = 50
$PSlot += $TTens
CellPhone_Glyph()
$PSlot = 50
$PSlot += $TSec
CellPhone_Glyph()
return
