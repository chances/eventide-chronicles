:Fae_801

        Player.LocateAnyMeans3D($PLAYER1, 602.3546, -1700.3068, 6.2431, 5.0, 5.0, 5.0, False)
        Player.IsInModel($PLAYER1, #SPEEDER)
    then
        if
            $SpeederDelivered == 0
        then
            Text.PrintHelp('SPEED')
            wait 5000
            goto @Fae_1000
        end
        if
            $Tool_Mission_Passed == 0
        then
            Text.PrintHelpString("Come back once you have completed ~g~Born of Dirt.")
            wait 5000
            goto @Fae_1000
        else
            goto @Fae_911
        end

end

  Player.CanStartMission($PLAYER1)
