:Water
script_name 'Water'
$WarningActive = 0

:Water_10
while Player.IsPlaying($PLAYER1)
    $InWater = 0
    $WaterCar = 0
    wait 0
    if
        Char.IsInWater($scplayer)
    then
        $InWater = 1
    else
        if
            Player.IsInAnyCar($PLAYER1)
        then
            Player.StoreCarIsInNoSave($PLAYER1, $Any_Car)
            if or
                Car.IsInWater($Any_Car)
                Car.IsDead($Any_Car)
            then
                $WaterCar = 1
            end
        else
            Player.GetCoordinates($PLAYER1, $PlayerX, $PlayerY, $PlayerZ)
            Path.GetClosestCharNode($PlayerX, $PlayerY, $PlayerZ, $SafeX, $SafeY, $SafeZ)
        end
    end

    if or
        $WaterCar == 1
        $InWater == 1
    then
        Text.PrintHelpForever('TPORT2')
        $WarningActive = 1
        if
            Pad.IsKeyPressed(KeyCode.N)
        then
            Text.ClearHelp()
            $WarningActive = 0
            if
                $WaterCar == 1
            then
                Camera.DoFade(500, Fade.Out)
                wait 500
                Player.SetControl($PLAYER1, False)
                Player.WarpFromCarToCoord($PLAYER1, $SafeX, $SafeY, $SafeZ)
                wait 500
                Camera.DoFade(500, Fade.In)
                wait 500
                Player.SetControl($PLAYER1, True)
                break
            else
                Camera.DoFade(500, Fade.Out)
                wait 500
                Player.SetControl($PLAYER1, False)
                Player.SetCoordinates($PLAYER1, $SafeX ,$SafeY, $SafeZ)
                wait 500
                Camera.DoFade(500, Fade.In)
                wait 500
                Player.SetControl($PLAYER1, True)
                break
            end
        end
    else
        if
            $WarningActive == 1
        then
            Text.ClearHelp()
            $WarningActive = 0
        end
    end

    if
        $Limey_Passed == 1
    then
        if
            $Barrier2Done == 1
        then
            Text.PrintHelpString("Aquatic teleportation network disabled. Please be wary of ~g~sharks ~w~and other ~g~large fish.")
            wait 5000
            Text.PrintHelpString("You can no longer survive in large bodies of water.")
            terminate_this_script
        end
    end
end

:Water_20
wait 0
if
    $WarningActive == 1
then
    Text.ClearHelp()
    $WarningActive = 0
end
goto @Water_10
