    then

    else

            Pickup.HasBeenCollected($BUSTED_HELP1)
        then
            goto @Help_400
        else
            if
                Pickup.HasBeenCollected($WASTED_HELP1)
            then
                goto @Help_682
            else
                if
                    Pickup.HasBeenCollected($BeachJettyInfo)
                then
                    goto @Help_777
                else
                    if or
                        Pickup.HasBeenCollected($BeachBribe1)
                        Pickup.HasBeenCollected($HavanaBribe)
                        Pickup.HasBeenCollected($BeachBribe2)
                        Pickup.HasBeenCollected($StarfishBribe)
                        Pickup.HasBeenCollected($ViceportBribe)
                        Pickup.HasBeenCollected($VicePointBribe)

                        goto @Help_3184
                    else
                        if
                            Pickup.HasBeenCollected($MarinaHelp)
                        then
                            goto @Help_3698
                        else
                            if
                                Pickup.HasBeenCollected($BarrierHelp)
                            then
                                goto @Help_4210
                            else
                                if
                                    $Barrier1 == 1
                                then
                                    goto @Help_5210
                                else
                                    if
                                        $Barrier2 == 1
                                    then
                                        goto @Help_5420
                                    else
                                        if
                                            Player.IsOnAnyBike($PLAYER1)
                                        then
                                            goto @Help_5600
                                        else
                                            if
                                                $PRINT_CAR_HELP == 0
                                            then
                                                goto @Help_5800
                                            else
                                                if and
                                                    $Barrier1Done == 1
                                                    $Barrier2Done == 1
                                                then
                                                    $HelpDone == 1
                                                    goto @Help_6000
                                                else
                                                    goto @Help_10
                                                end
                                            end
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
