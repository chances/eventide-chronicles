:MissionLock_Guard
terminate_this_script

/// Takes the mission lock if it is free.
/// @remarks
/// out: $MissionLockGranted = true when granted, false when busy. Read it
/// before the next `wait`. The test and the set have no `wait` between them.
/// The trace writes use `15@` as scratch; no other source uses `15@`.
:AcquireMissionLock
if
    $MissionLocked == false
then
    $MissionLocked = true
    $MissionLockGranted = true
    if DebugTrace
    then
        15@ = Clock.GetGameTimer()
        IniFile.WriteString("acquire", "trace.ini", "Trace", "lock")
        IniFile.WriteInt(15@, "trace.ini", "Trace", "lockTime")
    end
else
    $MissionLockGranted = false
end
return

/// Frees the mission lock. Any holder may call it; no owner is checked.
:ReleaseMissionLock
$MissionLocked = false
if DebugTrace
then
    15@ = Clock.GetGameTimer()
    IniFile.WriteString("release", "trace.ini", "Trace", "lock")
    IniFile.WriteInt(15@, "trace.ini", "Trace", "lockTime")
end
return
