:PACKAGE
script_name 'PACKAGE'

0@ = 0                                 // Active Target Package Index (0 to 87)
1@ = 1                                 // Blip Redraw Flag (1 = Needs New Blip)

:PACKAGE_Loop
while Player.IsPlaying($PLAYER1)
    wait 250

    // --- RADAR TRACKING LOGIC ---
    if 0@ <= 87
    then
        gosub @Get_Package_Coords

        // Draw/Redraw blip on active target package
        if 1@ == 1
        then
            1@ = 0
            Blip.Remove($CurrentPackBlip)
            $CurrentPackBlip = Blip.AddForCoord($PackCoordX, $PackCoordY, $PackCoordZ)
            Blip.ChangeColor($CurrentPackBlip, BlipColor.Red)
        end

        // PROXIMITY CHECK: Only advance 0@ IF player actually collects THIS target package
        if Player.LocateAnyMeans3D($PLAYER1, $PackCoordX, $PackCoordY, $PackCoordZ, 3.5, 3.5, 3.5, False)
        then
            0@ += 1                  // Target collected! Advance to next coordinate index
            1@ = 1                   // Signal blip redraw for the next package
            wait 1000                // Debounce pause so index doesn't double-increment
        end
    else
        // All package spots cleared
        Blip.Remove($CurrentPackBlip)
        terminate_this_script
    end
end

wait 250
goto @PACKAGE_Loop
